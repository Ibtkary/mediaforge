إدارة الصور بكفاءة،
وبنية واضحة

Mediaforge ليست مجرد أداة لرفع الصور، بل مكتبة مكتوبة بلغة Go لإدارة دورة حياة الصورة كاملة. تستقبل الصورة، تتحقق من سلامتها، تضغطها، تنشئ منها نسخًا بأحجام مناسبة، ثم ترفعها إلى التخزين السحابي وتعيد روابط مؤقتة وآمنة لعرضها.

$ go get github.com/Ibtkary/mediaforge

تحسين الأداء عبر WebP

تقليل حجم الصور واستهلاك النطاق الترددي مع الحفاظ على جودة مناسبة للعرض.

روابط آمنة ومؤقتة

توليد روابط موقعة لفترة محددة، حتى تُعرض الصور بأمان ودون كشف المسارات الداخلية.

مرونة مع مزودي التخزين

تصميم يدعم مزودات مثل AWS S3 وBunny.net دون تغيير منطق العمل في التطبيق.

رحلة تدفق الصورة

نظرة واضحة على ما يحدث خلف الكواليس، من لحظة استقبال الصورة حتى عرضها داخل التطبيق.

المرحلة الأولى

التحقق من الملف قبل المعالجة

فحص نوع الملف الحقيقيMagic Bytes validation

لا يعتمد النظام على امتداد الملف فقط، بل يفحص البايتات الأساسية للتأكد من أن الملف صورة فعلًا. وإذا كان الملف غير صالح أو يحمل محتوى مشبوهًا، تتوقف العملية مباشرة.

المرحلة الثانية

حساب البصمة وضغط الصورة

بصمة المحتوىContent Hash (SHA-256)

يُنشئ النظام بصمة فريدة للصورة بالاعتماد على محتواها الفعلي. وهذا يساعد على اكتشاف الصور المكررة وتجنب رفعها أكثر من مرة عند الحاجة.

التحويل إلى WebPImage Encoding Engine

بعد قراءة الصورة، يحولها النظام إلى صيغة WebP لتقليل الحجم وتحسين سرعة التحميل، خاصة في تطبيقات الهاتف والواجهات التي تعرض عددًا كبيرًا من الصور.

المرحلة الثالثة

إنشاء نسخ بأحجام مناسبة

النسخ متعددة الأحجامSmart Variants Generation

بدل إرسال صورة كبيرة لكل الحالات، ينشئ النظام عدة نسخ مناسبة لاحتياجات العرض المختلفة، مثل الصور المصغرة والقوائم وصفحات التفاصيل.

التثبيت والتهيئة

المكتبة مرنة من حيث مزود التخزين. يمكنك تثبيت الحزمة الأساسية ثم إضافة مزود التخزين المناسب لبيئتك.

المتطلبات الأساسية تحتاج المكتبة إلى Go 1.25+ على الأقل. كما تعتمد افتراضيًا على الأداة الخارجية cwebp لضغط الصور، لذلك يجب تثبيتها مسبقًا عبر brew install webp أو apt install webp.
سطر الأوامر 
# تثبيت الحزمة الأساسية
go get github.com/Ibtkary/mediaforge

# إذا كنت تستخدم AWS S3، ثبّت المزود الخاص به:
go get github.com/Ibtkary/mediaforge/providers/s3

خيارات التهيئة

عادةً تُنفذ التهيئة مرة واحدة عند بدء تشغيل الخادم، ثم يُمرر العميل الناتج إلى الخدمات والطبقات التي تحتاجه.

infrastructure/media/client.go 
func SetupBunnyEngine() (*mediaforge.Client, error) {
    return mediaforge.New(mediaforge.Config{
        StorageZone:    os.Getenv("BUNNY_ZONE"),
        StorageAPIKey:  os.Getenv("BUNNY_API_KEY"),
        CDNBaseURL:     os.Getenv("BUNNY_CDN_URL"),
        CDNSecurityKey: os.Getenv("BUNNY_CDN_KEY"),
    })
}
infrastructure/media/client.go 
import (
    "github.com/aws/aws-sdk-go-v2/config"
    "github.com/aws/aws-sdk-go-v2/service/s3"
    "github.com/Ibtkary/mediaforge"
    s3provider "github.com/Ibtkary/mediaforge/providers/s3"
)

func SetupS3Engine() (*mediaforge.Client, error) {
    awsCfg, _ := config.LoadDefaultConfig(context.Background())
    s3Client := s3.NewFromConfig(awsCfg)
    bucket := os.Getenv("S3_BUCKET")

    storage := s3provider.NewStorage(s3Client, bucket)
    signer := s3provider.NewSigner(s3.NewPresignClient(s3Client), bucket)

    return mediaforge.NewClient(storage, signer,
        mediaforge.WithMaxFileSize(10 * 1024 * 1024),
        mediaforge.WithVariants(mediaforge.DefaultVariants()),
    )
}

الدليل العملي للتكامل

عدد قليل من الدوال يكفي لإدارة دورة حياة الصورة: رفعها، عرضها، ثم حذفها عند الحاجة.

حدود مسؤولية المكتبة تتولى المكتبة الرفع والمعالجة وتوليد الروابط، لكنها لا تحفظ شيئًا في قاعدة البيانات. دورك هو أخذ نتيجة العملية وتخزين StoragePath والبيانات التي تحتاجها في جدولك.

1. رفع الصورة

// الرفع من []byte
asset, err := client.Upload(ctx, "tenant-id-or-store1", imageBytes)

// الرفع من io.Reader
asset, err := client.UploadReader(ctx, "store1", multipartStreamFile)

/* النتيجة ستكون PendingAsset يحتوي على:
   asset.StoragePath  // مسار الملف في التخزين
   asset.ContentHash  // بصمة المحتوى
   asset.Variants     // النسخ الناتجة للعرض
   بعد ذلك احفظ البيانات التي تحتاجها في قاعدة البيانات.
*/

2. جلب روابط العرض

// عندما يطلب التطبيق عرض الصورة:
urls, err := client.GetSignedURL(
    storagePath,                            // المسار المحفوظ في قاعدة البيانات
    []string{"thumb", "medium", "large"},   // النسخ المطلوبة
    2 * time.Hour,                          // مدة صلاحية الرابط
)

// ثم أعد الروابط للتطبيق:
response := map[string]string{
    "full_image_hd": urls.Original,
    "preview_icon":  urls.Variants["thumb"],
}

3. حذف الصورة

// عند حذف الصورة أو الكيان المرتبط بها:
results, err := client.Delete(ctx, "store1", "imagehash_xyz99.webp",
    []string{"thumb", "medium", "large"})

// تُحذف الصورة الأصلية والنسخ التابعة لها من التخزين.

النسخ المتعددة للصورة

إنشاء عدة نسخ من الصورة ليس أمرًا إضافيًا، بل خطوة مهمة لتحسين سرعة العرض وتقليل استهلاك البيانات.

النسخة الأبعاد المرجعية الحجم المتوقع متى تُستخدم
Original بنفس أبعاد المصدر، حتى الحد الأقصى المسموح 45KB - 100KB للشاشة الكاملة أو التكبير أو الحالات التي تحتاج أعلى دقة.
large بحد أقصى 800px عرضًا ~ 35KB مناسب لصفحات التفاصيل وعرض الصورة بحجم كبير نسبيًا.
medium بحد أقصى 400px عرضًا ~ 15KB مناسب لكروت المنتجات والشبكات والقوائم المتوسطة.
thumb بحد أقصى 150px عرضًا ~ 3KB للصور المصغرة جدًا مثل القوائم ونتائج البحث والسلة.

الإعدادات المتقدمة وتحسين الموارد

إذا كنت تحتاج إلى تقليل التكلفة أو ضبط سلوك المعالجة بشكل أدق، فهذه أهم الخيارات المتاحة.

1. منع التكرار (Dedup)

عند تفعيل WithDedup(true)، يتحقق النظام مما إذا كانت الصورة نفسها قد رُفعت من قبل. وإذا كانت موجودة، يمكنه تجاوز الرفع لتقليل النقل والتخزين.

2. التخطي الذكي (Smart Skip)

عند تفعيل WithSmartVariantSkip(true)، لا ينشئ النظام نسخًا كبيرة من صور صغيرة أصلًا، مما يقلل وقت المعالجة واستهلاك الموارد.

3. التوليد المؤجل (Lazy Generation)

مع WithLazyVariants(true) يمكن رفع الصورة الأصلية أولًا، ثم تأجيل إنشاء النسخ الأخرى إلى وقت لاحق إذا كان ذلك أنسب لتدفق العمل.

4. المراقبة والتتبع (Observability)

يمكنك استخدام WithHook(...) لربط المكتبة بأدوات المراقبة والتتبع، ومتابعة زمن العمليات والأخطاء دون خلط ذلك بمنطق التطبيق.

مرجع الواجهات البرمجية

ملخص لأهم الدوال المتاحة، والمدخلات التي تستقبلها، والقيم التي تعيدها.

الدوال الأساسية

الدالة المعاملات القيمة المرجعة الوصف
Upload ctx, tenantID, []byte *PendingAsset, error يرفع الصورة من الذاكرة، ويعالجها، ثم يخزنها ويعيد بياناتها.
UploadReader ctx, tenantID, io.Reader *PendingAsset, error يرفع الصورة من io.Reader بدل تحميلها كاملة في الذاكرة مسبقًا.
GetSignedURL storagePath, variants[], ttl *SignedURLSet, error ينشئ روابط موقعة ومؤقتة لعرض الصورة ونسخها بأمان.
Delete ctx, tenantID, filename, variants[] []DeleteResult, error يحذف الصورة الأصلية والنسخ التابعة لها ويعيد نتيجة كل عملية حذف.

فئات الأخطاء

تعيد المكتبة أخطاء منظمة من النوع *mediaforge.Error حتى يكون التعامل معها أوضح داخل التطبيق.

الفئة أمثلة للأسباب إعادة محاولة؟
CategoryValidation الصورة غير صالحة، أو أكبر من الحجم المسموح، أو نوعها غير مدعوم. لا
CategoryProcessing فشل أثناء معالجة الصورة أو أثناء التحويل إلى WebP. لا
CategoryStorage مشكلة مؤقتة في التخزين السحابي أو الشبكة. نعم، أحيانًا
